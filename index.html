async function login(){ const password = $('password').value.trim(); if(!password){ $('loginStatus').textContent = 'Password দিন'; return; } $('loginStatus').textContent = 'Checking...'; $('loginBtn').disabled = true; try{ const r = await fetch( '/api/auth', { method:'POST', headers:{ 'Content-Type': 'application/json' }, credentials:'same-origin', cache:'no-store', body:JSON.stringify({ password }) } ); const text = await r.text(); let d = {}; try{ d = JSON.parse(text); }catch{ throw new Error( 'Server returned invalid response.' ); } if(!r.ok){ throw new Error( d.error || 'Wrong password' ); } sessionStorage.setItem( 'nt_unlocked', '1' ); $('loginStatus').textContent = 'Unlocked ✓'; setTimeout(()=>{ $('login').classList.add('hidden'); },200); }catch(e){ console.error( 'LOGIN ERROR:', e ); $('loginStatus').textContent = e.message || 'Login failed.'; }finally{ $('loginBtn').disabled = false; } }
